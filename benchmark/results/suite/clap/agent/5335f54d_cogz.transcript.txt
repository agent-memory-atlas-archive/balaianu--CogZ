```rust
#[must_use]
#[cfg_attr(debug_assertions, track_caller)]
pub fn mut_subcommands<F>(mut self, f: F) -> Self
where
    F: FnMut(Command) -> Command,
{
    self.subcommands = self.subcommands.into_iter().map(f).collect();
    self
}
```
